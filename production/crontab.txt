# SOXL + A股 定时任务配置
# 安装方法: crontab production/crontab.txt
# 当前生效配置可直接用: crontab -l

# ============================================
# A股数据自动更新
# ============================================
# 每天 17:30 执行增量更新（A股收盘后）
30 17 * * 1-5 /bin/bash /home/rziiiii/dev/data/auto_update.sh

# 每周六凌晨 2:00 执行全量验证和导出
0 2 * * 6 /usr/bin/python3 /home/rziiiii/dev/data/main.py --validate

# ============================================
# SOXL Daily Scheduler (Asia/Shanghai local time)
# ============================================
# 21:00 中国时间 周一-五 — 美股盘前 cached-data signal + dashboard, no Futu order
0 21 * * 1-5 SOXL_SCHEDULE_TRIGGER=cron_preopen /Users/rziiiii/dev/workspace/production/run_soxl_scheduler.sh --broker futu --account-notional 10000 --runtime-composer budget_usage --composer-profile v19reg_v22_budget_usage_bull_guard_v1 --slow-profile balanced_rule_residual_v19_reg --fast-profile balanced_rule_residual_v22 --bull-participation-guard --dry-run --no-notify --skip-data-update --health-summary --schedule-slot preopen >> /Users/rziiiii/dev/workspace/runtime/soxl/paper_trading/scheduler_preopen.log 2>&1 || true

# 06:15 中国时间 周二-六 — 美股收盘后完整流程（数据 + 模拟盘 + dashboard）
15 6 * * 2-6 SOXL_SCHEDULE_TRIGGER=cron_postclose /Users/rziiiii/dev/workspace/production/run_soxl_scheduler.sh --broker futu --account-notional 10000 --runtime-composer budget_usage --composer-profile v19reg_v22_budget_usage_bull_guard_v1 --slow-profile balanced_rule_residual_v19_reg --fast-profile balanced_rule_residual_v22 --bull-participation-guard --no-notify --health-summary --schedule-slot postclose >> /Users/rziiiii/dev/workspace/runtime/soxl/paper_trading/scheduler_postclose.log 2>&1 || true

# ============================================
# SOXL 每周 Walk-Forward 验证
# ============================================
# 每周六 07:00 中国时间运行 walk-forward，排在收盘后数据刷新之后
0 7 * * 6 /Users/rziiiii/dev/workspace/production/run_soxl_weekly_wf.sh >> /Users/rziiiii/dev/workspace/runtime/soxl/wf_weekly.log 2>> /Users/rziiiii/dev/workspace/runtime/soxl/wf_weekly.err || true
