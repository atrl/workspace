# SOXL + A股 定时任务配置
# 安装方法: crontab production/crontab.txt
# 当前生效配置可直接用: crontab -l

# ============================================
# A股数据自动更新
# ============================================
# 每天 17:30 执行增量更新（A股收盘后）
30 17 * * 1-5 /bin/bash /home/rziiiii/dev/data/auto_update.sh

# 每周六凌晨 2:00 执行全量验证和导出
0 2 * * 6 /usr/bin/python3 /home/rziiiii/dev/data/main.py --validate

# ============================================
# SOXL Daily Scheduler (3次/工作日)
# ============================================
# 09:30 盘前 — 更新数据 + 生成交易信号
30 9 * * 1-5 cd /home/rziiiii/dev/workspace && qlib_venv/bin/python production/soxl_daily_scheduler.py --broker paper --no-notify --skip-dashboard >> runtime/soxl/paper_trading/scheduler_am.log 2>&1 || true

# 16:35 收盘后 — 完整流程（数据 + 交易 + dashboard）
35 16 * * 1-5 cd /home/rziiiii/dev/workspace && qlib_venv/bin/python production/soxl_daily_scheduler.py --broker paper --no-notify >> runtime/soxl/paper_trading/scheduler_pm.log 2>&1 || true

# 21:00 晚间 — 仅交易 + dashboard（跳过数据更新）
0 21 * * 1-5 cd /home/rziiiii/dev/workspace && qlib_venv/bin/python production/soxl_daily_scheduler.py --broker paper --no-notify --skip-data-update >> runtime/soxl/paper_trading/scheduler_eve.log 2>&1 || true

# ============================================
# SOXL 每周 Walk-Forward 验证
# ============================================
# 每周六凌晨 3:00 运行 walk-forward
0 3 * * 6 cd /home/rziiiii/dev/workspace && qlib_venv/bin/python soxl_qlib_strategy_v2.py --profiles balanced_trade_lock --walk-forward --seed 123 >> runtime/soxl/wf_weekly.log 2>&1 || true
